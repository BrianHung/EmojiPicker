## Usage

`EmojiNode` draws custom emoji, which are images, in the document. Without custom emoji, leave it out of `nodes`.

```tsx title="editor.tsx"
import { LexicalComposer } from "@lexical/react/LexicalComposer";
import { ContentEditable } from "@lexical/react/LexicalContentEditable";
import { LexicalErrorBoundary } from "@lexical/react/LexicalErrorBoundary";
import { RichTextPlugin } from "@lexical/react/LexicalRichTextPlugin";
import { EmojiProvider } from "@brianhung/emoji-picker-react";
import { EmojiNode } from "@/components/lexical/emoji-node";
import { EmojiTypeaheadPlugin } from "@/components/lexical/emoji-typeahead-plugin";

export function Editor() {
  return (
    <EmojiProvider>
      <LexicalComposer initialConfig={{ namespace: "editor", nodes: [EmojiNode], onError: console.error }}>
        <RichTextPlugin contentEditable={<ContentEditable />} ErrorBoundary={LexicalErrorBoundary} />
        <EmojiTypeaheadPlugin />
      </LexicalComposer>
    </EmojiProvider>
  );
}
```

## How it works

`useLexicalEmojiTypeahead` reads the `:query` before the caret, ranks the matches, and handles the keys through Lexical's command system. `EmojiSuggestions.Popover` positions the menu at the typed `:query`.

```tsx title="emoji-typeahead-plugin.tsx (excerpt)"
import { useLexicalEmojiTypeahead } from "@/hooks/emoji-lexical";

const emoji = useLexicalEmojiTypeahead({
  node: (text, { record, shortcode }) => (record.imageUrl ? $createEmojiNode({ unified: record.unified, shortcode, text: null }) : $createTextNode(`${text} `)),
});

<EmojiSuggestions.Popover open={emoji.open} anchor={emoji.anchor} onOpenChange={(open) => !open && emoji.dismiss()}>
  <ul {...emoji.suggestions.getListProps()}>
    {emoji.suggestions.suggestions.map((s, i) => <li key={s.shortcode} {...emoji.suggestions.getOptionProps(i)}>:{s.shortcode}:</li>)}
  </ul>
</EmojiSuggestions.Popover>
```
