## Usage

`EmojiNode` draws custom emoji, which are images, in the document. Without custom emoji, leave it out of `nodes`. The picker button needs the `emoji-picker-popover.json` item from [Installation](/#installation).

```tsx title="editor.tsx"
"use client";

import { $getRoot, $getSelection, $insertNodes, $isRangeSelection } from "lexical";
import { LexicalComposer } from "@lexical/react/LexicalComposer";
import { useLexicalComposerContext } from "@lexical/react/LexicalComposerContext";
import { ContentEditable } from "@lexical/react/LexicalContentEditable";
import { LexicalErrorBoundary } from "@lexical/react/LexicalErrorBoundary";
import { RichTextPlugin } from "@lexical/react/LexicalRichTextPlugin";
import { EmojiProvider, useEmojiStore } from "emoji-components";
import { EmojiPickerPopover } from "@/components/ui/emoji-picker-popover";
import { $createEmojiNode, EmojiNode, emojiAttrs } from "@/components/lexical/emoji-node";
import { EmojiSuggestionsPlugin } from "@/components/lexical/emoji-suggestions-plugin";

function InsertEmoji() {
  const [editor] = useLexicalComposerContext();
  const store = useEmojiStore();
  return (
    <EmojiPickerPopover
      aria-label="Insert emoji"
      onSelect={(selection) => {
        editor.focus();
        editor.update(() => {
          const range = $getSelection();
          const attrs = selection.isCustom ? emojiAttrs(store, selection.unified) : null;
          if (attrs) $insertNodes([$createEmojiNode(attrs)]);
          else ($isRangeSelection(range) ? range : $getRoot().selectEnd()).insertText(selection.emoji);
        });
      }}
    >
      🙂
    </EmojiPickerPopover>
  );
}

export function Editor() {
  return (
    <EmojiProvider>
      <LexicalComposer initialConfig={{ namespace: "editor", nodes: [EmojiNode], onError: console.error }}>
        <InsertEmoji />
        <RichTextPlugin contentEditable={<ContentEditable />} ErrorBoundary={LexicalErrorBoundary} />
        <EmojiSuggestionsPlugin />
      </LexicalComposer>
    </EmojiProvider>
  );
}
```

## How it works

`useLexicalEmojiSuggestions` reads the `:query` before the caret, ranks the matches, and handles the keys through Lexical's command system. The menu is Floating UI's: `anchor` is the `:query`'s box as a virtual element, and the hook keeps the keys, so the menu never takes focus.

```tsx title="emoji-suggestions-plugin.tsx (excerpt)"
import { autoUpdate, flip, FloatingPortal, offset, shift, useFloating } from "@floating-ui/react";
import { useLexicalEmojiSuggestions } from "@/hooks/use-emoji-lexical";

const emoji = useLexicalEmojiSuggestions({
  node: (text, { record, shortcode }) => (record.imageUrl ? $createEmojiNode({ unified: record.unified, shortcode, text: null }) : $createTextNode(`${text} `)),
});

const { refs, floatingStyles } = useFloating({
  open: emoji.open,
  placement: "bottom-start",
  whileElementsMounted: autoUpdate,
  middleware: [offset(6), flip(), shift({ padding: 8 })],
});
useLayoutEffect(() => refs.setPositionReference(emoji.anchor), [refs, emoji.anchor]);

emoji.open && (
  <FloatingPortal>
    <ul ref={refs.setFloating} style={floatingStyles} {...emoji.getListProps()}>
      {emoji.suggestions.map((s, i) => <li key={s.shortcode} {...emoji.getOptionProps(i)}>:{s.shortcode}:</li>)}
    </ul>
  </FloatingPortal>
);
```
