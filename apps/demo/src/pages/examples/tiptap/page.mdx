## Usage

```tsx title="editor.tsx"
import { EditorContent, useEditor } from "@tiptap/react";
import StarterKit from "@tiptap/starter-kit";
import { EmojiProvider } from "@brianhung/emoji-picker";
import { EmojiSuggestionMenu } from "./emoji-suggestion";

function Editor() {
  const editor = useEditor({ extensions: [StarterKit] });
  return (
    <EmojiProvider data="/emoji/en.json">
      <EditorContent editor={editor} />
      <EmojiSuggestionMenu editor={editor} />
    </EmojiProvider>
  );
}
```

## How it works

`useTiptapEmojiTypeahead` registers the library's ProseMirror plugin on the editor, ranks the matches and handles the keys, so the editor keeps its caret. `EmojiSuggestions.Popover` positions the menu at the typed `:query`. Copy [`emoji-suggestion.tsx`](https://github.com/BrianHung/EmojiPicker/blob/main/apps/demo/src/components/tiptap/emoji-suggestion.tsx) into your project.

```tsx title="emoji-suggestion.tsx (excerpt)"
const emoji = useTiptapEmojiTypeahead(editor, {
  content: (text, s) => (s.record.imageUrl ? [{ type: "emoji", attrs }, { type: "text", text: " " }] : `${text} `),
});

<EmojiSuggestions.Popover open={emoji.open} anchor={emoji.anchor} onOpenChange={(open) => !open && emoji.dismiss()}>
  <ul {...emoji.suggestions.getListProps()}>
    {emoji.suggestions.suggestions.map((s, i) => <li key={s.shortcode} {...emoji.suggestions.getOptionProps(i)}>:{s.shortcode}:</li>)}
  </ul>
</EmojiSuggestions.Popover>
```
