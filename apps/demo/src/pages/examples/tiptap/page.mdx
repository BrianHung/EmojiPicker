## Usage

`EmojiNode` draws custom emoji, which are images, in the document. Without custom emoji, leave it out. The picker button needs the `emoji-picker-popover.json` item from [Installation](/#installation).

```tsx title="editor.tsx"
"use client";

import { Suspense } from "react";
import { EditorContent, useEditor } from "@tiptap/react";
import StarterKit from "@tiptap/starter-kit";
import { EmojiProvider, useEmojiStore } from "emoji-components";
import { EmojiPickerPopover } from "@/components/ui/emoji-picker-popover";
import { EmojiNode, emojiAttrs } from "@/components/tiptap/emoji-node";
import { EmojiSuggestionsMenu } from "@/components/tiptap/emoji-suggestions-menu";

function TiptapEditor() {
  const store = useEmojiStore();
  const editor = useEditor({ extensions: [StarterKit, EmojiNode.configure({ store })], immediatelyRender: false });
  if (!editor) return null;
  return (
    <>
      <EmojiPickerPopover
        aria-label="Insert emoji"
        finalFocus={() => editor.view.dom}
        onSelect={(selection) => {
          const attrs = selection.isCustom ? emojiAttrs(store, selection.unified) : null;
          if (attrs) editor.chain().focus().insertEmoji(attrs).run();
          else editor.chain().focus().insertContent(selection.emoji).run();
        }}
      >
        🙂
      </EmojiPickerPopover>
      <EditorContent editor={editor} />
      <EmojiSuggestionsMenu editor={editor} />
    </>
  );
}

export function Editor() {
  return (
    <EmojiProvider>
      <Suspense>
        <TiptapEditor />
      </Suspense>
    </EmojiProvider>
  );
}
```

`immediatelyRender: false` renders the editor after hydration, as Tiptap requires with server rendering, and `<Suspense>` lets Next.js Cache Components prerender the page, since Tiptap calls `Math.random()` while it renders.

## How it works

`useTiptapEmojiSuggestions` registers the ProseMirror plugin on the editor. The plugin finds the `:query`, ranks the matches, and handles the keys. The menu is Floating UI's: `anchor` is the `:query`'s box as a virtual element, and the hook keeps the keys, so the menu never takes focus.

```tsx title="emoji-suggestions-menu.tsx (excerpt)"
import { autoUpdate, flip, FloatingPortal, offset, shift, useFloating } from "@floating-ui/react";
import { useTiptapEmojiSuggestions } from "@/hooks/use-emoji-tiptap";

const emoji = useTiptapEmojiSuggestions(editor, {
  content: (text, { record, shortcode }) => (record.imageUrl ? emojiContent({ unified: record.unified, shortcode, text: null }) : `${text} `),
});

const { refs, floatingStyles } = useFloating({
  open: emoji.open,
  placement: "bottom-start",
  whileElementsMounted: autoUpdate,
  middleware: [offset(6), flip(), shift({ padding: 8 })],
});
useLayoutEffect(() => refs.setPositionReference(emoji.anchor), [refs, emoji.anchor]);

emoji.open && (
  <FloatingPortal>
    <ul ref={refs.setFloating} style={floatingStyles} {...emoji.getListProps()}>
      {emoji.suggestions.map((s, i) => <li key={s.shortcode} {...emoji.getOptionProps(i)}>:{s.shortcode}:</li>)}
    </ul>
  </FloatingPortal>
);
```
