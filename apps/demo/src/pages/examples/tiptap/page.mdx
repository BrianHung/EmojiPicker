## Usage

`EmojiNode` draws custom emoji, which are images, in the document. Without custom emoji, leave it out.

```tsx title="editor.tsx"
import { EditorContent, useEditor } from "@tiptap/react";
import StarterKit from "@tiptap/starter-kit";
import { EmojiProvider, useEmojiStore } from "@brianhung/emoji-picker-react";
import { EmojiNode } from "@/components/tiptap/emoji-node";
import { EmojiSuggestionMenu } from "@/components/tiptap/emoji-suggestion";

function TiptapEditor() {
  const store = useEmojiStore();
  const editor = useEditor({ extensions: [StarterKit, EmojiNode.configure({ store })] });
  return (
    <>
      <EditorContent editor={editor} />
      <EmojiSuggestionMenu editor={editor} />
    </>
  );
}

export function Editor() {
  return (
    <EmojiProvider>
      <TiptapEditor />
    </EmojiProvider>
  );
}
```

## How it works

`useTiptapEmojiTypeahead` registers the ProseMirror plugin on the editor. The plugin finds the `:query`, ranks the matches, and handles the keys, so the editor keeps its caret. `EmojiSuggestions.Popover` positions the menu at the typed `:query`.

```tsx title="emoji-suggestion.tsx (excerpt)"
import { useTiptapEmojiTypeahead } from "@/hooks/emoji-tiptap";

const emoji = useTiptapEmojiTypeahead(editor, {
  content: (text, s) => (s.record.imageUrl ? [{ type: "emoji", attrs }, { type: "text", text: " " }] : `${text} `),
});

<EmojiSuggestions.Popover open={emoji.open} anchor={emoji.anchor} onOpenChange={(open) => !open && emoji.dismiss()}>
  <ul {...emoji.suggestions.getListProps()}>
    {emoji.suggestions.suggestions.map((s, i) => <li key={s.shortcode} {...emoji.suggestions.getOptionProps(i)}>:{s.shortcode}:</li>)}
  </ul>
</EmojiSuggestions.Popover>
```
